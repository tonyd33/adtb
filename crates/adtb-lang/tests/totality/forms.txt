# Cases of the totality search in adtb-lang's analysis tests.
#
# `row KEY` is the row of a table of docs/spec/prelude.md with a Faults column whose first cell,
# or first two cells joined by ` | ` where the first is not unique, is KEY; its texts are partial
# exactly when that cell names a fault, a fault `try` catches. `form FILE "PHRASE"` is a form the
# spec file names in PHRASE, and `expect total`, `expect fault` or `expect escape` states its class:
# an escape is partial and stays partial under `try`. `given` lists the parameters, `schema` adds a
# declaration, `hiding` judges as a bundle declaring `hide new constructors`, and `untried` marks a
# form `try` cannot enclose. Each indented line is one ad hoc text; one written `OP ~ TEXT` must
# reach the operator whose `Debug` form is OP, and each text of a row names the one its row names.

row `uuid (t: Text) -> Uuid`
given t: Text
  ParseUuid ~ uuid(t)

row `text (u: Uuid) -> Text`
given u: Uuid
  ToText ~ text(u)

row `a + b`, `a - b` | `Int -> Int -> Int`, one type
given n: I64
  Add ~ n + 1
  Sub ~ n - 1

row `a * b` | `Int -> Int -> Int`, one type
given n: I64

row `a / b`
given n: I64
  Div ~ 10 / n

row `a % b`
given n: I64

row `-a` | `Int -> Int`
given n: I64
  Neg ~ -n

row `a + b`, `a - b` | `Decimal s -> Decimal t -> Decimal (max(s, t))`
given d: Decimal 2
  Add ~ d + d
  Sub ~ d - d

row `a * b` | `Decimal s -> Decimal t -> Decimal (s + t)`; refused when `s + t > 38`
given d: Decimal 2
  Mul ~ d * d

row `-a` | `Decimal s -> Decimal s`
given d: Decimal 2
  Neg ~ -d

form prelude.md "Each checked form returns `None` exactly where its operator faults"
expect total
given n: I64, x: F64
  checked_add(n, 1)
  checked_sub(n, 1)
  checked_mul(n, 2)
  checked_div(10, n)
  checked_rem(10, n)
  checked_neg(n)
  checked_i32(n)
  checked_i64(x, HalfEven)

form prelude.md "the span; faults `InvalidSpan` when `start > end`"
expect fault
given a: I64, b: I64
  span(a, b)

form prelude.md "| `start [a: Ord] (s: Span a) -> a` | `s.start` |"
expect total

  overlaps(s, t)
  covers(s, t)
  covers(s, 3)

  p + p
  -p

  dt - d
