=== base auth
module auth
add module auth
add type Tier = Free | Paid
add type User = { id: Key users, name: Text, tier: Tier }
add relation users : User key id generate sequence
add type Identity { id: Key identities, note: Text }
  = EmailPass { email: Text }
  | Google { gid: Text }
add relation identities : Identity key id generate sequence
add type Price = { sku: Text, n: I64 }
add relation prices : Price key sku
add type Order = { id: Key orders, price: Key prices }
add relation orders : Order key id generate sequence
add type Pay = Card { last4: Text } | Cash
add type Payment { id: Key payments } = Paying { how: Pay } | Owed { amt: I64 }
add relation payments : Payment key id generate sequence
add type Scale = Realm | Town
add type Place { id: Key places } = Location { name: Text, scale: Scale } | Faction
add relation places : Place key id generate sequence

=== operator split
module auth
split constructor Place.Location into Region { name: Text } | Settlement { name: Text }
  by match
    | Location { name, scale: Realm } -> Region { name: name }
    | Location { name, scale: Town } -> Settlement { name: name }
--- inverse
module auth
add type Rs = Region | Settlement
merge constructor Place.(Region | Settlement) into Location { name: Text, rs: Rs }
--- blame
merge constructor Place.(Region | Settlement) into Location { name: Text, rs: Rs }
--- oracle
refusal
=== body split places
query places () -> Set Text = from p in places select match p | Region r -> r.name | Settlement t -> t.name | Faction _ -> "f"
