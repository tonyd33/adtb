=== relation tiered
module auth
add module auth
add type Tier = Free | Paid
add type P = { t: Tier, n: Text, w: Text }
add relation p : P key (t, n)
add type Q = { id: Key qs, o: Key p }
add relation qs : Q key id generate sequence
--- rows
insert p { t: Free, n: "a", w: "1" }
insert p { t: Free, n: "z", w: "9" }
insert qs { o: p.key(t: Free, n: "a") }
--- observe
from x in p select x.w

=== relation priced

add module auth
add type Price { sku: Text, n: I64, w: Text } = Std | Promo
add relation prices : Price key (sku, n)
add type Order = { id: Key orders, price: Key prices }
add relation orders : Order key id generate sequence
--- rows
insert prices.Std { sku: "a", n: 1, w: "1" }
insert prices.Promo { sku: "z", n: 9, w: "9" }
insert orders { price: prices.key(sku: "a", n: 1) }
--- observe
from x in prices select x.w
